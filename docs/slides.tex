\begin{frame}{研究背景：期待値最大化の限界と分布型強化学習}
  \begin{block}{強化学習の標準的定式化}
    方策 $\pi$ の下での割引リターンを確率変数 $Z^\pi(s,a)=\sum_{t\ge 0}\gamma^t R_t$ とみなし，
    その期待値 $Q^\pi(s,a)=\E[Z^\pi(s,a)]$ を最大化する．
  \end{block}
  \begin{alertblock}{期待値最大化では制御できないもの}
    破産・クラッシュのような\alert{発生確率は小さいが損失が大きい事象}は期待値にわずかしか寄与しない．
    期待値最大化方策はこの\alert{下側リスク}を回避する動機を持たず，
    リスクの評価・制御には $Z^\pi$ の\alert{分布そのもの}が必要である．
  \end{alertblock}
  \begin{block}{分布型強化学習（Distributional RL）}
    分布ベルマン作用素
    $\mathcal{T}^\pi Z(s,a) \overset{D}{=} R(s,a) + \gamma\, Z(S',A')$，
    $S'\sim P(\cdot\mid s,a)$，$A'\sim\pi(\cdot\mid S')$
    により $Z^\pi$ の分布を直接学習する．
    実装上は分位点関数 $\Finv(\tau)$（$\tau\in[0,1]$）を\blue{有限個 $N$ の原子}で離散近似する．
  \end{block}
\end{frame}

% ============================================================
% 本編 3: CVaR と下側裾
% ============================================================
\begin{frame}{目標リスク尺度：条件付きバリューアットリスク（CVaR）}
  \begin{columns}[onlytextwidth, T]
    \begin{column}{0.55\linewidth}
      \begin{block}{定義（Rockafellar \& Uryasev, 2000）}
        \small
        リターン $Z$ の分位点関数 $\Finv$，水準 $\alpha\in(0,1)$ に対し
        \[
          \CVaR_\alpha(Z) = \frac{1}{\alpha}\int_0^\alpha \Finv(\tau)\,d\tau
        \]
        すなわち\alert{最悪 $\alpha$ の事象に条件付けた期待リターン}．
        歪み関数 $g(\tau)=\min(\tau/\alpha,\,1)$ を用いると
        \[
          \CVaR_\alpha(Z)=\int_0^1 \Finv(\tau)\,g'(\tau)\,d\tau,\quad
          g'(\tau)=\tfrac{1}{\alpha}\ind[\tau<\alpha]
        \]
      \end{block}
    \end{column}
    \begin{column}{0.44\linewidth}
      \centering
      \begin{tikzpicture}[scale=0.74]
        \draw[->] (-3.4,0) -- (3.4,0) node[below]{\footnotesize $z$};
        \draw[->] (0,0) -- (0,1.55);
        \fill[red!30] plot[domain=-3.2:-1.645, samples=40] (\x,{1.157*exp(-\x*\x/2)}) -- (-1.645,0) -- (-3.2,0) -- cycle;
        \draw[thick, blue!70!black] plot[domain=-3.2:3.2, samples=80] (\x,{1.157*exp(-\x*\x/2)});
        \draw[dashed] (-1.645,0) -- (-1.645,0.48);
        \node[above right=-3pt] at (-1.645,0.38) {\footnotesize $\VaR_\alpha$};
        \node[red!70!black, below] at (-2.4,-0.02) {\footnotesize 確率 $\alpha$};
        \node[red!70!black, align=left] at (-3.1, 1.25) {\footnotesize $\CVaR_\alpha$：赤い領域の\\[-2pt] \footnotesize 条件付き期待値};
        \draw[->, red!70!black] (-2.2, 0.90) to[bend left=15] (-2.1, 0.22);
        \node[below] at (0,-0.35) {\footnotesize リターン分布の密度（概念図）};
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \begin{alertblock}{本質的な観察}
    $\CVaR_\alpha$ の推定精度は\alert{下側裾 $[0,\alpha]$ 上での $\Fhinv$ の近似精度のみ}で決まる．
    $\alpha=0.05$ なら残り $95\%$ の領域の近似精度は $\CVaR_\alpha$ に影響しない．
  \end{alertblock}
\end{frame}

% 本編 4: FQF と課題
% ============================================================
\begin{frame}{既存手法：FQF（Fully Parameterized Quantile Function）とその課題}
  \begin{columns}[T]
    \begin{column}{0.58\linewidth}
      \begin{block}{FQF（Yang et al., 2019）}
        \small
        \textbf{quantile fraction}（累積確率軸 $[0,1]$ の区切り）$0=\tau_0<\tau_1<\dots<\tau_N=1$ を
        FPN（Fraction Proposal Network）で状態依存に学習し，
        原子 $\theta_i=\Finv(\hat\tau_i)$，$\hat\tau_i=\tfrac{\tau_i+\tau_{i+1}}{2}$ で分布を表現する．
        目的関数は $W_1(Z,\hat Z)=\int_0^1|\Finv(\tau)-\Fhinv(\tau)|\,d\tau$ で，
        quantile fraction に関する勾配は閉形式
        \[
          \frac{\partial W_1}{\partial \tau_i}=2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1}).
        \]
      \end{block}
    \end{column}
    \begin{column}{0.40\linewidth}
      \centering
      \includegraphics[width=0.82\linewidth]{fig1_fqf_overview.png}\\
      {\scriptsize FQF の概要}
    \end{column}
  \end{columns}
  \begin{alertblock}{課題：一様重みによる解像度配分}
    $W_1$ は $[0,1]$ 上で\alert{一様に}誤差を評価するため，quantile fraction は確率質量に比例して配分される．
    下側裾 $[0,\alpha]$ に置かれる個数は概ね $\alpha N$（$\alpha=0.05,\ N=16$ で\alert{約 $0.8$ 個}）にとどまり，
    $\CVaR_\alpha$ の推定誤差が生じやすい．
  \end{alertblock}
\end{frame}

% 本編 5: 理論的着想
% ============================================================
\begin{frame}{理論的着想：CVaR推定誤差の上界と歪み重み付き目的関数}
  \begin{exampleblock}{命題（CVaR推定誤差の上界）}
    $g_\# Z$ を分位点関数 $\Finv\circ g^{-1}$ をもつ歪み分布とすると
    \[
      \bigl|\CVaR_\alpha(Z)-\CVaR_\alpha(\hat Z)\bigr|
      \le W_1\bigl(g_\# Z,\, g_\# \hat Z\bigr)
      = \int_0^1 g'(\tau)\,\bigl|\Finv(\tau)-\Fhinv(\tau)\bigr|\,d\tau =: \Lg(Z,\hat Z).
    \]
    {\footnotesize 証明の要点：$\E[g_\# Z]=\int_0^1\Finv(g^{-1}(u))\,du=\CVaR_\alpha(Z)$，
    $|\E X-\E Y|\le W_1(X,Y)$，変数変換 $u=g(\tau)$．}
  \end{exampleblock}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{block}{導かれる設計指針}
        FQF の目的関数 $W_1$ を $\Lg$ に置き換えれば，最小化対象が\blue{CVaR推定誤差の上界そのもの}になる．
        重み $g'(\tau)$ は下側裾のみを評価するので，quantile fraction は\blue{下側裾へ集中する}ことが最適となる．
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \centering
      \begin{tikzpicture}[xscale=4.4, yscale=0.9]
        \draw[->] (0,0.7) -- (1.03,0.7) node[right]{\scriptsize $\tau$};

\end{frame}

\begin{frame}{提案定理：quantile fraction に関する勾配の閉形式}
  \begin{exampleblock}{定理（歪み重み付き目的関数の境界勾配）}
    各区間 $[\tau_i,\tau_{i+1}]$ の原子を \textbf{$g$-中央値} $\hat\tau_i$，すなわち
    $g(\hat\tau_i)=\tfrac12\{g(\tau_i)+g(\tau_{i+1})\}$ にとる．このとき
    \[
      \frac{\partial \Lg}{\partial \tau_i}
      = \alert{g'(\tau_i)}\,\Bigl[\,2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1})\Bigr],
      \qquad i=1,\dots,N-1 .
    \]
  \end{exampleblock}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{block}{数理的・実装的な含意}
        \small
        \begin{itemize}
          \item FQF の勾配に\alert{$g'(\tau_i)$ を乗じるのみ}．ネットワーク構造の変更は不要．
          \item $g'(\tau_i)=1/\alpha$（裾内）は $\alpha=0.05$ で $20$ 倍，裾外は $0$．
                表現力が\blue{下側裾に自然に配分}される．
          \item 原子 $\theta_i$ の更新は分位点 $\hat\tau_i$ の分位点回帰で，従来と同一．
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{block}{アルゴリズム（歪みFQF）}
        \small
        \begin{enumerate}
          \item FPN が状態 $s$ から $\{\tau_i\}$ を出力
          \item 原子 $\theta_i(s,a)$ を分位点 $\hat\tau_i$ の分位点回帰で更新
          \item $\{\tau_i\}$ を上記の境界勾配で更新
          \item 行動選択：$a^\star=\argmax_a \widehat{\CVaR}_\alpha(Z(s,a))$
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ---------- A1 ----------
\begin{frame}{A1. quantile fraction に関する勾配の導出}
  \footnotesize
  \begin{block}{目的関数の分解}
    \[
      \Lg=\textstyle\sum_{j=0}^{N-1}\int_{\tau_j}^{\tau_{j+1}} g'(\tau)\,\bigl|\Finv(\tau)-\theta_j\bigr|\,d\tau,
      \qquad \theta_j=\Finv(\hat\tau_j),\quad g(\hat\tau_j)=\tfrac12\{g(\tau_j)+g(\tau_{j+1})\}.
    \]
    $\tau_i$ に依存する項は区間 $j=i-1$ と $j=i$ のみ．包絡項（$\theta_j$ 経由の依存）は
    $\theta_j$ が $g'$-重み付き $L^1$ 最小化の一階条件 $\int_{\tau_j}^{\hat\tau_j}g'=\int_{\hat\tau_j}^{\tau_{j+1}}g'$ を満たすため消える．
  \end{block}
  \begin{block}{ライプニッツ則の適用（$\Finv$ は非減少，$\hat\tau_{i-1}<\tau_i<\hat\tau_i$）}
    \begin{align*}
      \frac{\partial}{\partial\tau_i}\int_{\tau_{i-1}}^{\tau_i} g'\,|\Finv-\theta_{i-1}|\,d\tau
      &= g'(\tau_i)\bigl[\Finv(\tau_i)-\Finv(\hat\tau_{i-1})\bigr] + (\text{包絡項}=0), \\
      \frac{\partial}{\partial\tau_i}\int_{\tau_i}^{\tau_{i+1}} g'\,|\Finv-\theta_i|\,d\tau
      &= -\,g'(\tau_i)\bigl[\Finv(\hat\tau_i)-\Finv(\tau_i)\bigr] + 0 .
    \end{align*}
  \end{block}
  \begin{exampleblock}{境界項の和と結論}
    2つの境界項を加えて
    $\partial\Lg/\partial\tau_i=g'(\tau_i)\bigl[2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1})\bigr]$．
    注：$g'(\tau_i)=0$ の quantile fraction は $\Lg$ に直接寄与しないが，FPN の累積 softmax を通じて裾内の勾配が全ロジットに伝播する．
  \end{exampleblock}
\end{frame}

% ---------- A2 ----------
\begin{frame}{A2. 歪み分布の1次元最適量子化理論}
  \begin{block}{問題設定（Graf \& Luschgy, 2000）}
    確率測度 $\mu$（ここでは $g_\# Z$）を $N$ 点の離散測度で近似する最適 $N$-量子化問題：
    $e_{N,1}(\mu)=\inf\{W_1(\mu,\nu) : |\operatorname{supp}\nu|\le N\}$．
  \end{block}
  \begin{block}{最適解の特徴づけ（1次元・$L^1$）}
    \begin{itemize}
      \item 各区間 $[\tau_j,\tau_{j+1}]$ 上の最適原子は区間内の\alert{（重み付き）中央値}，
            すなわち $g'$-重みの下では $g$-中央値 $\hat\tau_j$ である．
      \item 最適な区切り $\tau_j$ は隣接原子の値の中点 $\tfrac12\{\theta_{j-1}+\theta_j\}$ に対応する（Voronoi 分割）．
            これは境界勾配 $2\Finv(\tau_j)-\theta_j-\theta_{j-1}=0$ と一致する．
      \item 量子化誤差は $e_{N,1}(\mu)\asymp N^{-1}$ で減衰する（Zador 型の漸近）．
    \end{itemize}
  \end{block}
  \begin{exampleblock}{本研究との対応}
    $\Lg$ の最小化は，$\tau$ 軸上で $g'$ を重みとする 1 次元最適量子化と同値である．
    実験A の log-log 傾き約 $-1.8$ は，$\alpha N\ge2$ の領域で理論的減衰以上の改善比 $R$ の伸びを示す．
  \end{exampleblock}
\end{frame}

% ---------- A3 ----------
\begin{frame}{A3. 実験Aの詳細設定}
  \footnotesize
  \begin{block}{対象分布（真の分位点関数が既知）}
    \begin{itemize}
      \item \textbf{D1 ガウス混合}：主成分（高確率）と負側の副成分（低確率・大損失）の2成分混合．クラッシュ型リターンを模擬．
      \item \textbf{D2 負の対数正規}：$Z=-Y$，$\log Y\sim\mathcal N(m,s^2)$．下側裾が重い非対称分布．
      \item \textbf{D3 Student-$t$（$\nu=3$）}：両側裾が重い対称分布．
    \end{itemize}
  \end{block}
  \begin{block}{手順}
    \begin{enumerate}
      \item 真の $\Finv$ から $\CVaR_\alpha(Z)=\tfrac1\alpha\int_0^\alpha\Finv$ を高精度数値積分（下側裾で細分化）で算出．
      \item 一様 $W_1$ と歪み $\Lg$ をそれぞれ $N=16$ 点で最小化．非凸性に備え\alert{マルチスタート}で最良解を採用．
      \item 各解の $\hat Z$ から誤差 $|\CVaR_\alpha(Z)-\CVaR_\alpha(\hat Z)|$ と改善比 $R$ を評価．
      \item $N$ を変えて点数換算値（歪み最適 $N$ 点と同誤差を達成する一様点数）を算出．
    \end{enumerate}
  \end{block}
  \begin{exampleblock}{補足結果}
    $\alpha N\ge 2$ の領域で最小 $R=42.3$，$N$ に対する $R$ の log-log 傾きは約 $-1.8$．
    歪み最適 $16$ 点 $\approx$ 一様 $695$ 点（$\alpha=0.05$）．
  \end{exampleblock}
\end{frame}

% ---------- A4 ----------
\begin{frame}{A4. 想定質問への回答（Q1〜Q3）}
  \begin{alertblock}{Q1. IQN の歪みサンプリングとの本質的な違いは？}
    IQN は $\tau\sim g^{-1}(U)$ のように\alert{読み出し側の確率操作}で歪みを扱い，分位点関数のパラメータ表現は $\tau$ 全域を対象とする．
    本研究は\alert{表現力（有限個の quantile fraction）そのものを下側裾に配分}する点で異なる．
  \end{alertblock}
  \begin{alertblock}{Q2. Beyond CVaR（Moghimi \& Ku, 2025）との違いは？}
    Beyond CVaR は\alert{固定}された quantile fraction の下で原子の\alert{値の重み付け}を行う．
    本研究は\alert{区切り位置 $\tau_i$ 自体を学習}し，配置の最適化を行う．両者は直交しており併用も可能である．
  \end{alertblock}
  \begin{alertblock}{Q3. 目的関数の重み付けは自明ではないか？}
    重み付け自体は自然な発想であるが，我々の調査した範囲では\alert{損失側で quantile fraction を歪み最適化する先行研究は見当たらない}．
    本研究の特徴は，CVaR推定誤差の 1-Wasserstein 上界 $\Lg$ から目的関数と勾配閉形式が\alert{自然に導出される}点にある．
  \end{alertblock}
\end{frame}

\begin{frame}{A5. 想定質問への回答（Q4〜Q5）}
  \begin{alertblock}{Q4. 実験規模（CPU 演算）の妥当性は？}
    実験B は Ryzen 7 5825U（CPU）で 1 run 約 35 分，100k ステップ・3 シード．
    RiskyChain-v0 は小規模環境であり，数千エピソードにわたる評価で\alert{統計的有意性を検証済み}（mean $\pm$ SD を報告）．
    大規模環境への拡張は今後の課題である．
  \end{alertblock}
  \begin{alertblock}{Q5. 実験B で IQN が MAE で優位だった理由と対策は？}
    \begin{itemize}
      \item 要因：FPN の\alert{非凸最適化}に起因する局所解と\alert{初期化バイアス}（初期配置が一様に近く，下側裾への移動が遅れる）．
            B1 の SD が大きい（$\pm0.75$）ことと整合する．
      \item 対策：デルタヘッジ環境では FPN の\alert{初期化改善}（$g$ に基づく初期配置）を適用する．
      \item なお B1 は FQF（B0）比で MAE を約 $1/5$ に低減しており，提案の主張（損失側の歪み重み付けの効果）は保持される．
    \end{itemize}
  \end{alertblock}
\end{frame}
