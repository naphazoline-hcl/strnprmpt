\begin{frame}{研究背景：期待値最大化の限界と分布型強化学習}
  \begin{block}{強化学習の標準的定式化}
    リターン $Z^\pi(s,a)=\sum_{t\ge 0}\gamma^t R_t$ の期待値 $Q^\pi(s,a)=\E[Z^\pi(s,a)]$ を最大化する．
  \end{block}
  \begin{alertblock}{期待値最大化では制御できないもの}
    破産・クラッシュのような\alert{稀な大損失}は期待値にほぼ寄与せず，
    \alert{下側リスク}を評価・制御できない．
  \end{alertblock}
  \begin{block}{分布型強化学習（Distributional RL）}
    $Z^\pi$ の\alert{分布そのもの}を学習．
    実装上は分位点関数 $\Finv(\tau)$（$\tau\in[0,1]$）を\blue{有限個 $N$ の原子}で近似する．
  \end{block}
\end{frame}

% ============================================================
% 本編 3: CVaR と下側裾（口頭: slides_script.md §3）
% ============================================================
\begin{frame}{目標リスク尺度：CVaR}
  \begin{columns}[onlytextwidth, T]
    \begin{column}{0.55\linewidth}
      \begin{block}{定義：最悪 $\alpha$ の事象に条件付けた期待リターン}
        \small
        \[
          \CVaR_\alpha(Z) = \frac{1}{\alpha}\int_0^\alpha \Finv(\tau)\,d\tau
        \]
        歪み関数 $g(\tau)=\min(\tau/\alpha,\,1)$ を用いると
        \[
          \CVaR_\alpha(Z)=\int_0^1 \Finv(\tau)\,g'(\tau)\,d\tau,\quad
          g'(\tau)=\tfrac{1}{\alpha}\ind[\tau<\alpha]
        \]
      \end{block}
    \end{column}
    \begin{column}{0.44\linewidth}
      \centering
      \begin{tikzpicture}[scale=0.74]
        \draw[->] (-3.4,0) -- (3.4,0) node[below]{\footnotesize $z$};
        \draw[->] (0,0) -- (0,1.55);
        \fill[red!30] plot[domain=-3.2:-1.645, samples=40] (\x,{1.157*exp(-\x*\x/2)}) -- (-1.645,0) -- (-3.2,0) -- cycle;
        \draw[thick, blue!70!black] plot[domain=-3.2:3.2, samples=80] (\x,{1.157*exp(-\x*\x/2)});
        \draw[dashed] (-1.645,0) -- (-1.645,0.48);
        \node[above right=-3pt] at (-1.645,0.38) {\footnotesize $\VaR_\alpha$};
        \node[red!70!black, below] at (-2.4,-0.02) {\footnotesize 確率 $\alpha$};
        \node[red!70!black, align=left] at (-3.1, 1.25) {\footnotesize $\CVaR_\alpha$：赤い領域の\\[-2pt] \footnotesize 条件付き期待値};
        \draw[->, red!70!black] (-2.2, 0.90) to[bend left=15] (-2.1, 0.22);
        \node[below] at (0,-0.35) {\footnotesize リターン分布の密度（概念図）};
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \begin{alertblock}{本質的な観察}
    $\CVaR_\alpha$ の推定精度は\alert{下側裾 $[0,\alpha]$ 上の近似精度のみ}で決まる
    （$\alpha=0.05$ なら残り $95\%$ の近似は無関係）．
  \end{alertblock}
\end{frame}

% 本編 4: FQF と課題（口頭: slides_script.md §4）
% ============================================================
\begin{frame}{既存手法：FQF とその課題}
  \begin{columns}[T]
    \begin{column}{0.58\linewidth}
      \begin{block}{FQF（Yang et al., 2019）}
        quantile fraction（累積確率軸の区切り）$0=\tau_0<\tau_1<\dots<\tau_N=1$ を
        FPN で学習し，原子 $\theta_i=\Finv(\hat\tau_i)$（$\hat\tau_i$：区間中点）で分布を表現．
        \[
          \frac{\partial W_1}{\partial \tau_i}=2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1}).
        \]
      \end{block}
    \end{column}
    \begin{column}{0.40\linewidth}
      \centering
      \includegraphics[width=0.82\linewidth]{fig1_fqf_overview.png}\\
      {\scriptsize FQF の概要}
    \end{column}
  \end{columns}
  \begin{alertblock}{課題：一様重みによる解像度配分}
    固定一様配置では下側裾 $[0,\alpha]$ に置かれる
    quantile fraction は\alert{およそ $\alpha N$ 個}（$\alpha=0.05,\ N=16$ で約 $0.8$ 個，
    実際には内部境界は\alert{$0$ 個}）．
    FQF は配置を学習するが，目的が一様重みのため\alert{裾への集中を直接には促さない}．
  \end{alertblock}
\end{frame}

% 本編 5: 理論的着想（口頭: slides_script.md §5）
% ============================================================
\begin{frame}{理論的着想：CVaR推定誤差の上界}
  \begin{exampleblock}{命題（CVaR推定誤差の上界）}
    \[
      \bigl|\CVaR_\alpha(Z)-\CVaR_\alpha(\hat Z)\bigr|
      \le W_1\bigl(\dist Z,\, \dist \hat Z\bigr)
      = \int_0^1 g'(\tau)\,\bigl|\Finv(\tau)-\Fhinv(\tau)\bigr|\,d\tau =: \Lg.
    \]
  \end{exampleblock}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{block}{導かれる設計指針}
        FQF の目的関数を $W_1\to\Lg$ に置換すれば，母集団設定では
        \blue{CVaR推定誤差の上界}を最小化する（実装は plug-in 擬似勾配を最適化）．
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \centering
      \begin{tikzpicture}[xscale=4.4, yscale=0.9]
        \draw[->] (0,0.7) -- (1.03,0.7) node[right]{\scriptsize $\tau$};

% 本編 6: 提案手法の核心（口頭: slides_script.md §6）

\begin{frame}{提案手法の核心：quantile fraction に関する勾配}
  \begin{exampleblock}{命題（母集団設定での歪み重み付き目的関数の境界勾配）}
    各区間の原子を \textbf{$g$-中央値} $\hat\tau_i$，すなわち
    $g(\hat\tau_i)=\tfrac12\{g(\tau_i)+g(\tau_{i+1})\}$ にとると
    \[
      \frac{\partial \Lg}{\partial \tau_i}
      = \alert{g'(\tau_i)}\,\Bigl[\,2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1})\Bigr],
      \qquad i=1,\dots,N-1 .
    \]
  \end{exampleblock}
  \begin{block}{数理的・実装的な含意}
    \begin{itemize}
      \item FQF の勾配に\alert{$g'(\tau_i)$ を乗じるのみ}．ネットワーク構造の変更は不要．
            実装はネットワーク出力で $\Finv$ を置換した \blue{plug-in 擬似勾配（stop-gradient）}．
      \item 裾内 $g'(\tau_i)\approx(1-\lambda)/\alpha+\lambda$（$\alpha=0.05,\lambda=0.05$ で約 $19$ 倍），
            裾外 $\lambda$ $\Rightarrow$ 表現力が\blue{下側裾に集中して配分}される
            （最適化には正則化 $g_{\alpha,\lambda}$ を使用）．
    \end{itemize}
  \end{block}
\end{frame}

      \textbf{本研究} & \alert{損失側（歪み重み $\Lg$）} & \green{学習} & \alert{$\Lg$ を直接最小化} \\
      \bottomrule
    \end{tabular}
  \end{block}

  \end{exampleblock}
\end{frame}

  真の分位点関数が既知の3分布（D1 ガウス混合，D2 負の対数正規，D3 Student-$t$）で，
  一様 $W_1$ と歪み $\Lg$ の $N=16$ 点配置を比較．
  \begin{columns}[onlytextwidth, T]

    \begin{column}{0.26\linewidth}

      {\footnotesize CVaR 推定誤差}
    \end{column}
    \begin{column}{0.39\linewidth}
      \begin{exampleblock}{主結果（改善比 $R=$ 一様誤差／歪み最適誤差）}
        \footnotesize
        \centering
        \begin{tabular}{@{}lccc@{}}
          \toprule
          分布 & 一様 & 歪み最適 & $R$ \\
          \midrule
          D1 & $0.309$ & $0.00316$ & $\mathbf{97.9}$ \\

          \item 経験的傾き：M2 \alert{$-1.8$}（二次減衰），M0 $\alert{-0.72}$〜$-1.27$（一次減衰）
        \end{itemize}
        \par\vspace{1pt}{\scriptsize ※$N=16$・$\alpha=0.05$ は一様配置が単一分位点のみで CVaR を推定する特殊設定．}
      \end{exampleblock}

  \end{columns}
\end{frame}
