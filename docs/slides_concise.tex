\begin{frame}{研究背景：期待値最大化の限界と分布型強化学習}
  \begin{block}{強化学習の標準的定式化}
    リターン $Z^\pi(s,a)=\sum_{t\ge 0}\gamma^t R_t$ の期待値 $Q^\pi(s,a)=\E[Z^\pi(s,a)]$ を最大化する．
  \end{block}
  \begin{alertblock}{期待値最大化では制御できないもの}
    破産・クラッシュのような\alert{稀な大損失}は期待値にほぼ寄与せず，
    \alert{下側リスク}を評価・制御できない．
  \end{alertblock}
  \begin{block}{分布型強化学習（Distributional RL）}
    $Z^\pi$ の\alert{分布そのもの}を学習．
    実装上は分位点関数 $\Finv(\tau)$（$\tau\in[0,1]$）を\blue{有限個 $N$ の原子}で近似する．
  \end{block}
\end{frame}

% ============================================================
% 本編 3: CVaR と下側裾（口頭: slides_script.md §3）
% ============================================================
\begin{frame}{目標リスク尺度：CVaR}
  \begin{columns}[onlytextwidth, T]
    \begin{column}{0.55\linewidth}
      \begin{block}{定義：最悪 $\alpha$ の事象に条件付けた期待リターン}
        \small
        \[
          \CVaR_\alpha(Z) = \frac{1}{\alpha}\int_0^\alpha \Finv(\tau)\,d\tau
        \]
        歪み関数 $g(\tau)=\min(\tau/\alpha,\,1)$ を用いると
        \[
          \CVaR_\alpha(Z)=\int_0^1 \Finv(\tau)\,g'(\tau)\,d\tau,\quad
          g'(\tau)=\tfrac{1}{\alpha}\ind[\tau<\alpha]
        \]
      \end{block}
    \end{column}
    \begin{column}{0.44\linewidth}
      \centering
      \begin{tikzpicture}[scale=0.74]
        \draw[->] (-3.4,0) -- (3.4,0) node[below]{\footnotesize $z$};
        \draw[->] (0,0) -- (0,1.55);
        \fill[red!30] plot[domain=-3.2:-1.645, samples=40] (\x,{1.157*exp(-\x*\x/2)}) -- (-1.645,0) -- (-3.2,0) -- cycle;
        \draw[thick, blue!70!black] plot[domain=-3.2:3.2, samples=80] (\x,{1.157*exp(-\x*\x/2)});
        \draw[dashed] (-1.645,0) -- (-1.645,0.48);
        \node[above right=-3pt] at (-1.645,0.38) {\footnotesize $\VaR_\alpha$};
        \node[red!70!black, below] at (-2.4,-0.02) {\footnotesize 確率 $\alpha$};
        \node[red!70!black, align=left] at (-3.1, 1.25) {\footnotesize $\CVaR_\alpha$：赤い領域の\\[-2pt] \footnotesize 条件付き期待値};
        \draw[->, red!70!black] (-2.2, 0.90) to[bend left=15] (-2.1, 0.22);
        \node[below] at (0,-0.35) {\footnotesize リターン分布の密度（概念図）};
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \begin{alertblock}{本質的な観察}
    $\CVaR_\alpha$ の推定精度は\alert{下側裾 $[0,\alpha]$ 上の近似精度のみ}で決まる
    （$\alpha=0.05$ なら残り $95\%$ の近似は無関係）．
  \end{alertblock}
\end{frame}

\begin{frame}{既存手法：FQF とその課題}
  \begin{columns}[T]
    \begin{column}{0.58\linewidth}
      \begin{block}{FQF（Yang et al., 2019）}
        quantile fraction（累積確率軸の区切り）$0=\tau_0<\tau_1<\dots<\tau_N=1$ を
        FPN で学習し，原子 $\theta_i=\Finv(\hat\tau_i)$（$\hat\tau_i$：区間中点）で分布を表現．
        \[
          \frac{\partial W_1}{\partial \tau_i}=2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1}).
        \]
      \end{block}
    \end{column}
    \begin{column}{0.40\linewidth}
      \centering
      \includegraphics[width=0.82\linewidth]{fig1_fqf_overview.png}\\
      {\scriptsize FQF の概要}
    \end{column}
  \end{columns}
  \begin{alertblock}{課題：一様重みによる解像度配分}
    固定一様配置では下側裾 $[0,\alpha]$ に置かれる
    quantile fraction は\alert{およそ $\alpha N$ 個}（$\alpha=0.05,\ N=16$ で約 $0.8$ 個，
    実際には内部境界は\alert{$0$ 個}）．
    FQF は配置を学習するが，目的が一様重みのため\alert{裾への集中を直接には促さない}．
  \end{alertblock}
\end{frame}

\begin{frame}{提案手法の核心：quantile fraction に関する勾配}
  \begin{exampleblock}{命題（母集団設定での歪み重み付き目的関数の境界勾配）}
    各区間の原子を \textbf{$g$-中央値} $\hat\tau_i$，すなわち
    $g(\hat\tau_i)=\tfrac12\{g(\tau_i)+g(\tau_{i+1})\}$ にとると
    \[
      \frac{\partial \Lg}{\partial \tau_i}
      = \alert{g'(\tau_i)}\,\Bigl[\,2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1})\Bigr],
      \qquad i=1,\dots,N-1 .
    \]
  \end{exampleblock}
  \begin{block}{数理的・実装的な含意}
    \begin{itemize}
      \item FQF の勾配に\alert{$g'(\tau_i)$ を乗じるのみ}．ネットワーク構造の変更は不要．
            実装はネットワーク出力で $\Finv$ を置換した \blue{plug-in 擬似勾配（stop-gradient）}．
      \item 裾内 $g'(\tau_i)\approx(1-\lambda)/\alpha+\lambda$（$\alpha=0.05,\lambda=0.05$ で約 $19$ 倍），
            裾外 $\lambda$ $\Rightarrow$ 表現力が\blue{下側裾に集中して配分}される
            （最適化には正則化 $g_{\alpha,\lambda}$ を使用）．
    \end{itemize}
  \end{block}
\end{frame}

% ---------- A1 ----------
\begin{frame}{A1. quantile fraction に関する勾配の導出}
  \footnotesize
  \begin{block}{目的関数の分解}
    \[
      \Lg=\textstyle\sum_{j=0}^{N-1}\int_{\tau_j}^{\tau_{j+1}} g'(\tau)\,\bigl|\Finv(\tau)-\theta_j\bigr|\,d\tau,
      \qquad \theta_j=\Finv(\hat\tau_j),\quad g(\hat\tau_j)=\tfrac12\{g(\tau_j)+g(\tau_{j+1})\}.
    \]
    $\tau_i$ に依存する項は区間 $j=i-1$ と $j=i$ のみ．包絡項（$\theta_j$ 経由の依存）は
    $\theta_j$ が $g'$-重み付き $L^1$ 最小化の一階条件 $\int_{\tau_j}^{\hat\tau_j}g'=\int_{\hat\tau_j}^{\tau_{j+1}}g'$ を満たすため消える．
  \end{block}
  \begin{block}{ライプニッツ則の適用（$\Finv$ は非減少，$\hat\tau_{i-1}<\tau_i<\hat\tau_i$）}
    \begin{align*}
      \frac{\partial}{\partial\tau_i}\int_{\tau_{i-1}}^{\tau_i} g'\,|\Finv-\theta_{i-1}|\,d\tau
      &= g'(\tau_i)\bigl[\Finv(\tau_i)-\Finv(\hat\tau_{i-1})\bigr] + (\text{包絡項}=0), \\
      \frac{\partial}{\partial\tau_i}\int_{\tau_i}^{\tau_{i+1}} g'\,|\Finv-\theta_i|\,d\tau
      &= -\,g'(\tau_i)\bigl[\Finv(\hat\tau_i)-\Finv(\tau_i)\bigr] + 0 .
    \end{align*}
  \end{block}
  \begin{exampleblock}{境界項の和と結論}
    2つの境界項を加えて
    $\partial\Lg/\partial\tau_i=g'(\tau_i)\bigl[2\Finv(\tau_i)-\Finv(\hat\tau_i)-\Finv(\hat\tau_{i-1})\bigr]$．
    注：実装は正則化歪み $g_{\alpha,\lambda}$（$\lambda=0.05$）で裾外にも $g'=\lambda>0$ が流れ，累積 softmax 経由でも裾内の勾配が伝播する．
  \end{exampleblock}
\end{frame}

% ---------- A2 ----------
\begin{frame}{A2. 歪み分布の1次元最適量子化理論}
  \small
  \begin{block}{問題設定（Graf \& Luschgy, 2000）}
    確率測度 $\mu$（ここでは歪み分布 $\dist Z$）を $N$ 点の離散測度で近似する最適 $N$-量子化問題：
    $e_{N,1}(\mu)=\inf\{W_1(\mu,\nu) : |\operatorname{supp}\nu|\le N\}$．
  \end{block}
  \begin{block}{最適解の特徴づけ（1次元・$L^1$）}
    \begin{itemize}
      \item 各区間 $[\tau_j,\tau_{j+1}]$ 上の最適原子は区間内の\alert{（重み付き）中央値}，
            すなわち $g'$-重みの下では $g$-中央値 $\hat\tau_j$ である．
      \item 最適な区切り $\tau_j$ は隣接原子の値の中点 $\tfrac12\{\theta_{j-1}+\theta_j\}$ に対応する（Voronoi 分割）．
            これは境界勾配 $2\Finv(\tau_j)-\theta_j-\theta_{j-1}=0$ と一致する．
      \item 量子化誤差は $e_{N,1}(\mu)\asymp N^{-1}$ で減衰する（Zador 型の漸近）．
    \end{itemize}
  \end{block}
  \begin{exampleblock}{本研究との対応と収束レート}
    \footnotesize
    $\Lg$ の最小化は，歪み分布 $\dist Z$ の 1 次元最適量子化と同値である．
    実験A の経験的傾き（歪み最適 $-1.79$〜$-1.84$，一様 $-0.72$〜$-1.27$）は，一様格子が裾特異性に律速され適応配置が回避する機序（本文 注意4.1）と整合する．
    補足：D3 では M0 誤差の 83〜95\%（$N\ge16$）が第1区間 $[0,1/N]$ に由来．適応配置の第1区間幅は，点密度 $p\propto((\Finv)')^{1/2}\propto\tau^{-2/3}$（$t_3$）を $\int_0^{\tau_1}p\,d\tau=1/N$ に代入して $\tau_1\propto N^{-3}$（実測 $0.4\,N^{-3}$）．
  \end{exampleblock}
\end{frame}

% ---------- A3 ----------
\begin{frame}{A3. 実験Aの詳細設定}
  \footnotesize
  \begin{block}{対象分布（裾形状・非対称性の異なる3分布，真の分位点関数が既知）}
    \begin{itemize}
      \item \textbf{D1 ガウス混合}：$0.9\,\mathcal N(0,1)+0.1\,\mathcal N(-4,1)$．クラッシュ型リターンを模擬．
      \item \textbf{D2 負の対数正規}：$Z=e^{\sigma^2/2}-e^{\sigma\xi}$，$\xi\sim\mathcal N(0,1)$，$\sigma=1$．下側裾が重い非対称分布．
      \item \textbf{D3 Student-$t$（$\nu=3$）}：両側裾が重い対称分布．
    \end{itemize}
  \end{block}
  \begin{block}{手順}
    \begin{enumerate}
      \item 真の $\Finv$ と部分期待値の閉形式から $\CVaR_\alpha(Z)$ を高精度に算出（数値反転の許容誤差 $10^{-12}$，MC クロスチェック）．
      \item 一様（M0），$g$-一様（M1），数値最適化（M2，マルチスタート L-BFGS-B．大域最適性は保証しない），Lloyd 型固定点（M3，クロスチェック）を比較．
      \item 誤差 $|\CVaR_\alpha(Z)-\CVaR_\alpha(\hat Z)|$ と改善比 $R$ を評価．$N\in\{8,\dots,1024\}$ に拡張測定．
      \item 点数換算値（歪み最適 $N$ 点と同誤差を達成する一様点数）を二分探索で算出．
    \end{enumerate}
  \end{block}
  \begin{exampleblock}{補足結果}
    $\alpha N\ge 2$ の領域で最小 $R=42.3$．誤差の経験的 log--log 傾きは歪み最適 $-1.79$〜$-1.84$（$R^2\ge0.998$），
    一様 $-0.72$〜$-1.27$（$\alpha=0.05$，$N=8$〜$1024$ の併合データ）．
    歪み最適 $16$ 点 $\approx$ 一様 $695$ 点（D1・$\alpha=0.05$；D2・D3 は 1024 点でも未達）．
  \end{exampleblock}
\end{frame}

% ---------- A4 ----------
\begin{frame}{A4. 想定質問への回答（Q1〜Q3）}
  \begin{alertblock}{Q1. IQN の歪みサンプリングとの本質的な違いは？}
    IQN は $\tau\sim g^{-1}(U)$ のように\alert{読み出し側の確率操作}で歪みを扱い，分位点関数のパラメータ表現は $\tau$ 全域を対象とする．
    本研究は\alert{表現力（有限個の quantile fraction）そのものを下側裾に配分}する点で異なる．
  \end{alertblock}
  \begin{alertblock}{Q2. Beyond CVaR（Moghimi \& Ku, 2025）との違いは？}
    Beyond CVaR は\alert{固定}された quantile fraction の下で原子の\alert{値の重み付け}を行う．
    本研究は\alert{区切り位置 $\tau_i$ 自体を学習}し，配置の最適化を行う．両者は直交しており併用も可能である．
  \end{alertblock}
  \begin{alertblock}{Q3. 目的関数の重み付けは自明ではないか？}
    重み付け自体は自然な発想であるが，我々の調査した範囲では\alert{損失側で quantile fraction を歪み最適化する先行研究は見当たらない}．
    本研究の特徴は，CVaR推定誤差の 1-Wasserstein 上界 $\Lg$ から目的関数と勾配閉形式が\alert{自然に導出される}点にある．
  \end{alertblock}
\end{frame}

\begin{frame}{A5. 想定質問への回答（Q4〜Q5）}
  \begin{alertblock}{Q4. 実験規模（CPU 演算）の妥当性は？}
    実験B は Ryzen 7 5825U（CPU）で 1 run 約 35 分，100k ステップ・3 シード（標本SD, ddof=1）．
    3 シードでは有意性検定は\alert{未実施}．シード $10$ 以上と bootstrap 信頼区間は今後の課題．
  \end{alertblock}
  \begin{alertblock}{Q5. 実験B で IQN が MAE で優位だった理由と対策は？}
    \scriptsize
    \begin{itemize}
      \item 理由1：IQN は評価時の読み出しで $(0,\alpha)$ 内に\alert{等間隔 16 点}を割り当てられ，裾に約 14 点の提案法より読み出し構造が有利．
      \item 理由2：M-est の参照値は各手法の greedy 方策下の分布であり，\alert{推定対象の難易度差}が混在（B0：重い裾，B1g/B2：ほぼ点質量）＋toy 環境の単純さ．
      \item 対抗手法 B2p（学習時歪み $\tau$）は MAE $3.42$ と約 10 倍悪化（\alert{ターゲット側の歪みサンプリングによる系統的バイアス}の解釈）．ただし sampling 系一般の劣位を示すものではなく，公平比較には\alert{重要度補正あり・現在側のみ・ターゲット側のみ}の対照が必要（今後の課題）．
    \end{itemize}
  \end{alertblock}
\end{frame}
