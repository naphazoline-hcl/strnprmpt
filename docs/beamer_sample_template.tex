  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{リスク感応型強化学習とCVaR}
        \begin{itemize}
          \item 収益分布の下側 $\alpha$ 部分の平均である \alert{CVaR} を最適化基準とする。
          \item CVaR の推定精度は\textbf{下側裾の近似精度}に支配される：
          \vspace{-4pt}
          \[
            \CVaR_\alpha(Z) = \frac{1}{\alpha}\int_0^{\alpha} \Finv(\tau)\,d\tau
          \]
          \vspace{-6pt}
        \end{itemize}
      \end{block}
      
      \begin{alertblock}{既存手法の課題（問題意識）}
        \begin{itemize}
          \item 有限個の原子で近似すると、下側裾 $[0,\alpha]$ の原子が不足する。
          \item ヘビーテイル損失を持つ環境では、この近似誤差が\red{ヘッジ方策を破綻}させる。
        \end{itemize}
      \end{alertblock}
    \end{column}

\begin{frame}{提案手法：歪み重み付き1-Wasserstein距離による学習}
  \begin{block}{歪み重み付き1-Wasserstein損失 $L_g$}
    {\small
    真の分位点関数 $\Finv$ に対し、歪み関数の微分 $g'$ で重み付けした $W_1$ 損失を最小化する：
    \vspace{-4pt}
    \[
      L_g(\tau_{1:N-1}) = \sum_{i=0}^{N-1}\int_{\tau_i}^{\tau_{i+1}} g'(\tau)\bigl|\Finv(\tau) - \Finv(\hat{\tau}_i)\bigr|\,d\tau
    \]
    \vspace{-6pt}
    }
  \end{block}

  \begin{exampleblock}{Proposition（quantile fractionに関する勾配）}
    {\small
    原子を $g$-中央値 $\hat{\tau}_i$（$g(\hat{\tau}_i) = \frac{g(\tau_i)+g(\tau_{i+1})}{2}$）に置くとき、
    境界 $\tau_i$ に関する勾配は\green{簡潔な解析形}となる：
    \vspace{-4pt}
    \[
      \frac{\partial L_g}{\partial \tau_i} = \alert{g'(\tau_i)}\bigl[2\Finv(\tau_i) - \Finv(\hat{\tau}_i) - \Finv(\hat{\tau}_{i-1})\bigr]
    \]
    \vspace{-6pt}
    }
  \end{exampleblock}

  \begin{itemize}\setlength{\itemsep}{2pt}
    \item FQF原論文の一様重み勾配に対して \alert{$g'(\tau_i)$ 倍}するのみで実装可能（最小限のコード改変）。
    \item CVaR（$g'(\tau) = 1/\alpha \cdot \mathbb{I}\{\tau \le \alpha\}$）では勾配が下側裾 $[0,\alpha]$ の原子を強力に牽引。
  \end{itemize}
\end{frame}
